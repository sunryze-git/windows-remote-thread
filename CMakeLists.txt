cmake_minimum_required(VERSION 3.20)
project(windows_remote_thread LANGUAGES CXX)

add_executable(remote_thread main.cpp)
target_compile_features(remote_thread PRIVATE cxx_std_17)
target_compile_definitions(remote_thread PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(remote_thread PRIVATE advapi32)

add_library(payload SHARED payload.cpp)
target_compile_features(payload PRIVATE cxx_std_17)
target_compile_definitions(payload PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(payload PRIVATE user32)
set_target_properties(payload PROPERTIES PREFIX "" OUTPUT_NAME "payload")

add_library(audio_payload SHARED audio_payload.cpp)
target_compile_features(audio_payload PRIVATE cxx_std_17)
target_compile_definitions(audio_payload PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(audio_payload PRIVATE mfplat mf mfuuid ole32 shlwapi uuid)
set_target_properties(audio_payload PROPERTIES PREFIX "" OUTPUT_NAME "audio_payload")

if (MINGW)
	target_link_options(remote_thread PRIVATE -municode -static)
	target_link_options(payload PRIVATE -static)
	target_link_options(audio_payload PRIVATE -static)
endif()

if (MSVC)
	target_compile_options(remote_thread PRIVATE /W4 /permissive-)
	target_compile_options(payload PRIVATE /W4 /permissive-)
	target_compile_options(audio_payload PRIVATE /W4 /permissive-)
else()
	target_compile_options(remote_thread PRIVATE -Wall -Wextra -Wpedantic)
	target_compile_options(payload PRIVATE -Wall -Wextra -Wpedantic)
	target_compile_options(audio_payload PRIVATE -Wall -Wextra -Wpedantic)
endif()
